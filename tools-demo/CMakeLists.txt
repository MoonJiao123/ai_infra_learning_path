cmake_minimum_required(VERSION 3.16)
project(cpp_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(demo main.cpp)
target_compile_options(demo PRIVATE -Wall -Wextra -g)
option(SANITIZE "Enable AddressSanitizer and UBSan" OFF)
if(SANITIZE)
 target_compile_options(demo PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(demo PRIVATE -fsanitize=address,undefined)
endif()
